-- Creating Professors Table
CREATE TABLE Professors (
    ProfessorID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    FirstName VARCHAR2(50) NOT NULL,
    LastName VARCHAR2(50) NOT NULL,
    Email VARCHAR2(100) UNIQUE NOT NULL  
);

-- Creating Courses Table
CREATE TABLE Courses (
    CourseID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    CourseName VARCHAR2(100) NOT NULL,
    Credits NUMBER(2) CHECK (Credits > 0),
    ProfessorID NUMBER NOT NULL,
    FOREIGN KEY (ProfessorID) REFERENCES Professors(ProfessorID) ON DELETE CASCADE
);

-- Creating Students Table 
CREATE TABLE Students (
    StudentID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    FirstName VARCHAR2(50) NOT NULL,
    LastName VARCHAR2(50) NOT NULL,
    Email VARCHAR2(100) UNIQUE NOT NULL,
    DOB DATE NOT NULL,
    Program VARCHAR2(100) NOT NULL,  
    StudentLevel VARCHAR2(20) NOT NULL CHECK (StudentLevel IN ('Undergraduate', 'Postgraduate'))
);

-- Creating Enrollments Table 
CREATE TABLE Enrollments (
    EnrollmentID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    StudentID NUMBER NOT NULL,
    CourseID NUMBER NOT NULL,
    Grade CHAR(2) CHECK (Grade IN ('A', 'B', 'C', 'D','F')),
    FOREIGN KEY (StudentID) REFERENCES Students(StudentID) ON DELETE CASCADE,
    FOREIGN KEY (CourseID) REFERENCES Courses(CourseID) ON DELETE CASCADE
);

-- Creating Attendance Table
CREATE TABLE Attendance (
    AttendanceID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    EnrollmentID NUMBER NOT NULL,
    AttendanceDate DATE NOT NULL,
    AttendanceStatus VARCHAR2(10) NOT NULL CHECK (AttendanceStatus IN ('Present', 'Absent')),
    FOREIGN KEY (EnrollmentID) REFERENCES Enrollments(EnrollmentID) ON DELETE CASCADE
);

-- SQL Queries
-- Query 1: Find courses with more than 5 students enrolled
SELECT c.CourseID, c.CourseName, COUNT(e.StudentID) AS EnrollmentCount
FROM Courses c
JOIN Enrollments e ON c.CourseID = e.CourseID
GROUP BY c.CourseID, c.CourseName
HAVING COUNT(e.StudentID) > 5;

-- Query 2: Find professors who are teaching more than one course
SELECT p.ProfessorID, p.FirstName, p.LastName, COUNT(c.CourseID) AS TotalCourses
FROM Professors p
JOIN Courses c ON p.ProfessorID = c.ProfessorID
GROUP BY p.ProfessorID, p.FirstName, p.LastName
HAVING COUNT(c.CourseID) > 1;

-- Query 3: Identify professors whose courses have the highest total student enrollments
SELECT p.ProfessorID, p.FirstName, p.LastName, COUNT(e.EnrollmentID) AS Total_Enrollments
FROM Professors p
JOIN Courses c ON p.ProfessorID = c.ProfessorID
JOIN Enrollments e ON c.CourseID = e.CourseID
GROUP BY p.ProfessorID, p.FirstName, p.LastName;

-- Query 4: Find students who are enrolled in "Database Systems"
SELECT s.StudentID, s.FirstName, s.LastName 
FROM Students s
WHERE EXISTS (
    SELECT 1 FROM Enrollments e
    JOIN Courses c ON e.CourseID = c.CourseID
    WHERE e.StudentID = s.StudentID AND c.CourseName = 'Database Systems'
);

-- Query 5: Find students who have never been marked as absent
SELECT s.StudentID, s.FirstName, s.LastName 
FROM Students s
WHERE NOT EXISTS (
    SELECT 1 FROM Enrollments e
    JOIN Attendance a ON e.EnrollmentID = a.EnrollmentID
    WHERE e.StudentID = s.StudentID AND a.AttendanceStatus = 'Absent'
);

-- Query 6: Find students who are enrolled in the most number of courses
SELECT StudentID, FirstName, LastName, CourseCount
FROM (
    SELECT s.StudentID, s.FirstName, s.LastName, COUNT(e.CourseID) AS CourseCount,
           RANK() OVER (ORDER BY COUNT(e.CourseID) DESC) AS Rank
    FROM Students s
    JOIN Enrollments e ON s.StudentID = e.StudentID
    GROUP BY s.StudentID, s.FirstName, s.LastName
) WHERE Rank = 1;

-- Query 7: Find courses where the average grade is below 2.5 (on a 4.0 scale)
SELECT CourseID, CourseName, AvgGrade FROM (
    SELECT c.CourseID, c.CourseName,
           AVG(CASE 
                WHEN e.Grade = 'A' THEN 4 
                WHEN e.Grade = 'B' THEN 3 
                WHEN e.Grade = 'C' THEN 2 
                WHEN e.Grade = 'D' THEN 1 
                ELSE 0 
           END) AS AvgGrade
    FROM Courses c
    JOIN Enrollments e ON c.CourseID = e.CourseID
    GROUP BY c.CourseID, c.CourseName
) WHERE AvgGrade < 2.5;

-- Query 8: Identify the top 3 professors whose courses have the highest average student grade
SELECT p.ProfessorID, p.FirstName, p.LastName, 
       AVG(CASE 
            WHEN e.Grade = 'A' THEN 4 
            WHEN e.Grade = 'B' THEN 3 
            WHEN e.Grade = 'C' THEN 2 
            WHEN e.Grade = 'D' THEN 1 
            ELSE 0 
       END) AS AvgGrade
FROM Professors p
JOIN Courses c ON p.ProfessorID = c.ProfessorID
JOIN Enrollments e ON c.CourseID = e.CourseID
GROUP BY p.ProfessorID, p.FirstName, p.LastName
ORDER BY AvgGrade DESC FETCH FIRST 3 ROWS ONLY;

-- Query 9: Find the courses with the highest and lowest enrollment count
SELECT CourseID, CourseName, Total_Enrollments FROM (
    SELECT c.CourseID, c.CourseName, COUNT(e.EnrollmentID) AS Total_Enrollments,
           RANK() OVER (ORDER BY COUNT(e.EnrollmentID) DESC) AS HighRank,
           RANK() OVER (ORDER BY COUNT(e.EnrollmentID) ASC) AS LowRank
    FROM Courses c
    JOIN Enrollments e ON c.CourseID = e.CourseID
    GROUP BY c.CourseID, c.CourseName
) WHERE HighRank = 1 OR LowRank = 1;

-- Query 10: Find courses where students perform best and worst
SELECT CourseID, CourseName, MedianGrade, HighRank, LowRank
FROM (
    SELECT c.CourseID, c.CourseName,
           PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY 
                CASE 
                    WHEN e.Grade = 'A' THEN 4 
                    WHEN e.Grade = 'B' THEN 3 
                    WHEN e.Grade = 'C' THEN 2 
                    WHEN e.Grade = 'D' THEN 1 
                    ELSE 0 
                END) AS MedianGrade,
           RANK() OVER (ORDER BY PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY 
                CASE 
                    WHEN e.Grade = 'A' THEN 4 
                    WHEN e.Grade = 'B' THEN 3 
                    WHEN e.Grade = 'C' THEN 2 
                    WHEN e.Grade = 'D' THEN 1 
                    ELSE 0 
                END) DESC) AS HighRank,
           RANK() OVER (ORDER BY PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY 
                CASE 
                    WHEN e.Grade = 'A' THEN 4 
                    WHEN e.Grade = 'B' THEN 3 
                    WHEN e.Grade = 'C' THEN 2 
                    WHEN e.Grade = 'D' THEN 1 
                    ELSE 0 
                END) ASC) AS LowRank
    FROM Courses c
    JOIN Enrollments e ON c.CourseID = e.CourseID
    GROUP BY c.CourseID, c.CourseName
) WHERE HighRank = 1 OR LowRank = 1;
